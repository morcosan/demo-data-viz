import { Meta } from '@storybook/addon-docs/blocks'
import { CLASS_PREFIX, TOKENS } from '@ds/core'
import { DocsCard, DocsCanvas, DocsToken, DocsTokenCodeGrid, DocsTokenGrid } from '@ds/docs/core'
import { kebabCase } from 'lodash-es'
import { Fragment } from 'react'

export const tokens = Object.entries(TOKENS.SURFACE)

export const Surface = ({ title, children, className }) => (
  <div
    className={cx(
      'py-xs-7 px-sm-0 min-h-md-0 gap-xs-5 text-size-sm flex h-fit w-full flex-col justify-center',
      className,
    )}
  >
    <div>{title}</div>
    {children}
  </div>
)

export const TokenValue = ({ value }) => {
  return <DocsToken value={value} type={typeof value === 'string' && value.startsWith('oklch') ? 'color' : undefined} />
}

export const gridClass = cx('gap-x-xs-4 gap-y-xs-4 my-xs-4 grid w-max grid-cols-[auto_1fr] items-center')

export const parseRef = (ref) => {
  return typeof ref === 'string' && ref.startsWith('{') && ref.endsWith('}') ? ref.slice(1, -1) : ref
}

# Surface tokens

  <Surface title="Page" className="ds-surface-page">
    <Surface title="Sheet" className="ds-surface-sheet">
      <Surface title="Card" className="ds-surface-card">
        <Surface title="Popup" className="ds-surface-popup" />
      </Surface>
    </Surface>
    <Surface title="Card" className="ds-surface-card mt-xs-5">
      <Surface title="Popup" className="ds-surface-popup" />
    </Surface>
    <Surface title="Popup" className="ds-surface-popup mt-xs-5" />
  </Surface>
  <div className="gap-xs-9 flex flex-col">
    <div className="gap-xs-9 flex">
      <Surface title="Card" className="ds-surface-card" />
      <Surface title="Card Hover" className="ds-surface-card-hovered" />
      <Surface title="Card + Hover" className="ds-surface-card hover:ds-surface-card-hovered" />
    </div>
    <div className="gap-xs-9 flex">
      <Surface title="Button Solid" className="ds-surface-button-solid" />
      <Surface title="Button Solid Hover" className="ds-surface-button-solid-hovered" />
      <Surface title="Button Solid + Hover" className="ds-surface-button-solid hover:ds-surface-button-solid-hovered" />
    </div>
    <div className="gap-xs-9 flex">
      <Surface title="Button Danger" className="ds-surface-button-danger" />
      <Surface title="Button Danger Hover" className="ds-surface-button-danger-hovered" />
      <Surface
        title="Button Danger + Hover"
        className="ds-surface-button-danger hover:ds-surface-button-danger-hovered"
      />
    </div>
    <div className="gap-xs-9 flex">
      <Surface title="Button Caution" className="ds-surface-button-caution" />
      <Surface title="Button Caution Hover" className="ds-surface-button-caution-hovered" />
      <Surface
        title="Button Caution + Hover"
        className="ds-surface-button-caution hover:ds-surface-button-caution-hovered"
      />

<DocsCard>
  <table className="docs">
    <thead>
      <tr>
        <th>Token</th>
        <th>Reference</th>
        <th className="w-full">Value</th>
        <th>Coding</th>
      </tr>
    </thead>
    <tbody>
      {tokens.map(([name, token]) => (
        <tr key={name}>
          <td>
            <pre>{name}</pre>
          </td>
          <td>
            <div className={gridClass}>
              {Object.entries(token.ref).map(([key, value]) => (
                <Fragment key={key}>
                  <pre className="text-color-text-subtle">{kebabCase(key)}:</pre>
                  {typeof value === 'object' && !Array.isArray(value) ? (
                    <DocsTokenGrid
                      lightSlot={<DocsToken value={parseRef(value.light)} />}
                      darkSlot={<DocsToken value={parseRef(value.dark)} />}
                      large={Array.isArray(value.light) && Array.isArray(value.dark)}
                    />
                  ) : (
                    <DocsToken value={parseRef(value)} />
                  )}
                </Fragment>
              ))}
            </div>
          </td>
          <td className="pr-0!">
            <div className={gridClass}>
              {Object.entries(token.value).map(([key, value]) => (
                <Fragment key={key}>
                  <pre className="text-color-text-subtle">{kebabCase(key)}:</pre>
                  {typeof value === 'object' && !Array.isArray(value) ? (
                    <DocsTokenGrid
                      lightSlot={<TokenValue value={value.light} />}
                      darkSlot={<TokenValue value={value.dark} />}
                      large={Array.isArray(value.light) && Array.isArray(value.dark)}
                    />
                  ) : (
                    <DocsToken value={value} />
                  )}
                </Fragment>
              ))}
            </div>
          </td>
          <td>
            <DocsTokenCodeGrid
              tsVar={`tokens.color['${name}']`}
              classVar={`${CLASS_PREFIX.SURFACE}${name}`}
              tsSize="w-lg-2"
              classSize="w-lg-2"
              delay={1000}
              vertical
            />
          </td>
        </tr>
      ))}
    </tbody>
  </table>
</DocsCard>
